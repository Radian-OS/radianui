## Create Project

Run the `init` command to create a new Vite project:

<br />
<PackageManagerTabs
	commands={{
		npm: "npx radianui@latest init",
		yarn: "yarn dlx radianui@latest init",
		pnpm: "pnpm dlx radianui@latest init",
		bun: "bunx radianui@latest init",
	}}
/>

## Add Components

You can now start adding components to your project.

<br />
<PackageManagerTabs
	commands={{
		npm: "npx radianui@latest add button",
		yarn: "yarn dlx radianui@latest add button",
		pnpm: "pnpm dlx radianui@latest add button",
		bun: "bunx radianui@latest add button",
	}}
/>
<br />
The command above will add the `Button` component to your project. You can then import it like this:
<br/>
<CodeSnippet
					title="app.tsx"
					showLineNumber
					code={`import {Button} from "@/components/ui/Button"

export default function App() {
  return (
    <div>
        <Button variant="strong" />
            Button
        </Button>         
    </div>
  )
}                   `}
/>

### Edit tsconfig.json file

The current version of Vite splits TypeScript configuration into three files, two of which need to be edited. Add the baseUrl and paths properties to the compilerOptions section of
the tsconfig.json and tsconfig.app.json files:

<br />
<CodeSnippet
	title="tsconfig.json"
	showLineNumber
	code={`
{
		"files": [],
		"references": [
			{
			"path": "./tsconfig.app.json"
			},
			{
			"path": "./tsconfig.node.json"
			}
		],
		"compilerOptions": {
			"baseUrl": ".",
			"paths": {
				"@/*": ["./src/*"]
			}
		}
}`}
/>

### Edit tsconfig.app.json file

<CodeSnippet
	title="tsconfig.app.json"
	showLineNumber
	code={`
{
		"compilerOptions": {
			// ...
			"baseUrl": ".",
			"paths": {
				"@/*": [
					"./src/*"
				]
			}
			// ...
		}
}`}
/>

### Update vite.config.ts

Add the following code to the vite.config.ts so your app can resolve paths without error:

<br />
<PackageManagerTabs
	commands={{
		npm: "npm install -D @types/node",
		yarn: "yarn add -D @types/node",
		pnpm: "pnpm add -D @types/node",
		bun: "bun add -D @types/node",
	}}
/>
<br/>
 <CodeSnippet
	title="vite.config.ts"
	showLineNumber
	code={
`import path from "path"
import tailwindcss from "@tailwindcss/vite"
import react from "@vitejs/plugin-react"
import { defineConfig } from "vite"

// https://vite.dev/config/
export default defineConfig({
plugins: [react(), tailwindcss()],
resolve: {
alias: {
"@": path.resolve(\_\_dirname, "./src"),
},
},
})`}
/>
