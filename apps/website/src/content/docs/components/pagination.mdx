## Usage

### Imports

<CodeSnippet title="/components/ui/pagination" code={`import { Pagination, PaginationContent, PaginationEllipsis, PaginationItem } from "@/components/ui/pagination"`} />

### Code

<CodeSnippet
	title="/components/ui/pagination"
	code={`<Pagination>

<PaginationContent>

<PaginationItem>
	<Link href="#">First</Link>
</PaginationItem>

<PaginationItem>
	<PaginationEllipsis />
</PaginationItem>

<PaginationItem>
	<Link href="#">Second</Link>
</PaginationItem>

    </PaginationContent>

</Pagination>`}
/>
