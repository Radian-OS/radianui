## Usage

### Imports

```tsx showLineNumbers title="/components/ui/pagination"
import {
	Pagination,
	PaginationContent,
	PaginationEllipsis,
	PaginationItem,
} from "@/components/ui/pagination"
```

### Code

```tsx showLineNumbers title="/components/ui/pagination"
<Pagination>
	<PaginationContent>
		<PaginationItem>
			<Link href="#">First</Link>
		</PaginationItem>
		<PaginationItem>
			<PaginationEllipsis />
		</PaginationItem>
		<PaginationItem>
			<Link href="#">Second</Link>
		</PaginationItem>
	</PaginationContent>
</Pagination>
```
