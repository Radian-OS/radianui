## Installation

    	<Steps>
        <Step>Install the following dependencies:</Step>

        <Tabs defaultValue="cli">
          <TabsContent value="cli" className="p-0">
            <PackageManagerTabs
              commands={{
                pnpm: "pnpm i sonner",
                npm: "npm i sonner",
                yarn: "yarn i sonner",
                bun: "bunx i sonner",
            }}
            />
        </TabsContent>

        </Tabs>

        <Step>Wrap your application with the `Toaster` component from `sonner`.</Step>

        <CodeSnippet
        	title="app/layout.tsx"
        	code={`import { Toaster } from "@/components/ui/sonner"

export default function RootLayout({ children }) {
    return (
      <html lang="en">
        <head />
        <body>
          <main>{children}</main>
          <Toaster />
        </body>
      </html>
    )
  }
          `}
        />

        <Step>Update the import paths as per your project structure.</Step>

    </Steps>

### Imports

	title="Usage"

	    variant="outline"
	    onClick={() =>
	    	toast("Toast Title", {
	    		description: "Toast Description Message",
	    		action: {
	    			label: "Upgrade",
	    			onClick: () => console.log("Upgrade clicked!"),
	    		},
	    	})
	    }>
	    Show Toast
</Button>`}
/>
