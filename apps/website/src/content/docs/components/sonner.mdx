## Installation

    	<Steps>
        <Step>Install the following dependencies:</Step>

        <Tabs defaultValue="cli">
          <TabsContent value="cli" className="p-0">
          ```bash
          npm install sonner
          ```
        </TabsContent>

        </Tabs>

        <Step>Wrap your application with the `Toaster` component from `sonner`.</Step>

        ```tsx showLineNumbers title="app/layout.tsx"
        import { Toaster } from "sonner"

        export default function RootLayout({ children }) {
          return (
            <html lang="en">
              <head />
              <body>
                <main>{children}</main>
                <Toaster />
              </body>
            </html>
          )
        }
        ```

        <Step>Update the import paths as per your project structure.</Step>

    </Steps>

## Usage

### Imports

```tsx showLineNumbers title="/components/ui/sonner.tsx"
import { toast } from "sonner"
```

### Code

```tsx showLineNumbers title="/components/ui/sonner.tsx"
<Button
	variant="outline"
	onClick={() =>
		toast("Toast Title", {
			description: "Toast Description Message",
			action: {
				label: "Upgrade",
				onClick: () => console.log("Upgrade clicked!"),
			},
		})
	}>
	Show Toast
</Button>
```
