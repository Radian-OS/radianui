## Usage

### Imports

<CodeSnippet title="/components/ui/sonner" code={`import { toast } from "sonner"`} />

### Code

<CodeSnippet
	title="Usage"
	code={`<Button
	    variant="outline"
	    onClick={() =>
	    	toast("Toast Title", {
	    		description: "Toast Description Message",
	    		action: {
	    			label: "Upgrade",
	    			onClick: () => console.log("Upgrade clicked!"),
	    		},
	    	})
	    }>
	    Show Toast
</Button>`}
/>
