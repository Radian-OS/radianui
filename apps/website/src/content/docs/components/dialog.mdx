### Imports

<CodeSnippet
	title="/components/ui/modal"
	code={`import { 
		Dialog, 
		DialogClose, 
		DialogContent, 
		DialogDescription, 
		DialogFooter, 
		DialogHeader, 
		DialogTitle, 
		DialogTrigger 
    } from "@/components/ui/modal";`}
/>

### Code

<CodeSnippet title="/components/ui/modal" code={`<Dialog>

      <DialogTrigger asChild>
        <button className="px-4 py-2 bg-blue-500 text-white rounded">Open Dialog</button>
      </DialogTrigger>

      <DialogContent>
        <DialogHeader>
          <DialogTitle>My Dialog</DialogTitle>
          <DialogClose />
        </DialogHeader>

        <DialogBody>
          <p>This is a simple modal body content.</p>
        </DialogBody>

        <DialogFooter>
          <button className="px-3 py-1 bg-gray-200 rounded">Cancel</button>
          <button className="px-3 py-1 bg-blue-500 text-white rounded">Confirm</button>
        </DialogFooter>
      </DialogContent>

</Dialog>`} />
