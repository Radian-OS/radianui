## Usage

### Imports

<CodeSnippet
	language="tsx"
	title="/components/ui/drawer.tsx"
	code={`import { Drawer,
    DrawerBody,
    DrawerClose,
    DrawerContent,
    DrawerDescription,
    DrawerFooter,
    DrawerHeader,
    DrawerTitle,
    DrawerTrigger } from "@/registry/ui/drawer"`}
/>

### Code

<CodeSnippet
	language="tsx"
	title="/components/ui/drawer.tsx"
	code={`<Drawer handle={false} backdrop='overlay'>
		<DrawerTrigger><Button>Open Trigger</Button></DrawerTrigger>
		<DrawerContent>
			<Button variant="outline" color="neutral" >Close Drawer</Button>
		</DrawerContent>
</Drawer>`}
/>
