## Usage

### Imports

```tsx showLineNumbers title="/components/ui/carousel.tsx"
import {
	Carousel,
	CarouselContent,
	CarouselPrevious,
} from "@/components/ui/carousel"
```

### Code

```tsx showLineNumbers title="/components/ui/carousel.tsx"
<Carousel>
	<CarouselContent>
		<div>Slide 1</div>
		<div>Slide 2</div>
		<div>Slide 3</div>
	</CarouselContent>
	<CarouselPrevious>Previous</CarouselPrevious>
</Carousel>
```
