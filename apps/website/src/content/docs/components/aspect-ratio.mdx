## Usage

### Imports

```tsx showLineNumbers title="/components/ui/aspect-ratio.tsx"
import { AspectRatio } from "@/components/ui/aspect-ratio"
```

### Code

```tsx showLineNumbers title="/components/ui/aspect-ratio.tsx"
<AspectRatio ratio={16 / 9}>
	<Image src="/media/background-2.png" alt="Example" fill />
</AspectRatio>
```
