## Usage

### Imports

<CodeSnippet title="/components/ui/breadcrumb.tsx" code={`import { Breadcrumb, BreadcrumbItem } from "@/components/ui/breadcrumb"`} />

### Code

<CodeSnippet title="/components/ui/breadcrumb.tsx" code={`<Breadcrumb separator="slash">

        <BreadcrumbItem asChild>
          <Link href="/">Home</Link>
        </BreadcrumbItem>

        <BreadcrumbItem asChild>
          <Link href="/products">Products</Link>
        </BreadcrumbItem>

        <BreadcrumbItem asChild>
          <Link href="/products/electronics">Electronics</Link>
        </BreadcrumbItem>

        <BreadcrumbItem isCurrent>Headphones</BreadcrumbItem>

</Breadcrumb>`} />
