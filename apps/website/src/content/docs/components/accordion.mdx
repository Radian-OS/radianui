## Usage

### Imports

<CodeSnippet title="/components/ui/accordion.tsx" code={`import { Accordion, AccordionContent, AccordionItem, AccordionTrigger } from "@/components/ui/accordion"`} />

### Code

<CodeSnippet title="/components/ui/accordion.tsx" code={`<Accordion size="sm" variant="box" interaction="single" className="max-w-md mx-auto">

      <AccordionItem value="item-1">
        <AccordionTrigger>First Item</AccordionTrigger>
        <AccordionContent>
          <p>This is the content of the first accordion item.</p>
        </AccordionContent>
      </AccordionItem>

      <AccordionItem value="item-2">
        <AccordionTrigger>Second Item</AccordionTrigger>
        <AccordionContent>
          <p>This is the content of the second accordion item.</p>
        </AccordionContent>
      </AccordionItem>

</Accordion>`}/>
