## Usage

### Imports

```tsx showLineNumbers title="/components/ui/accordion.tsx"
import {
	Accordion,
	AccordionContent,
	AccordionItem,
	AccordionTrigger,
} from "@/components/ui/accordion"
```

### Code

```tsx showLineNumbers title="/components/ui/accordion.tsx"
<Accordion type="single" collapsible>
	<AccordionItem value="item-1">
		<AccordionTrigger>First Item</AccordionTrigger>
		<AccordionContent>
			<p>This is the content of the first accordion item.</p>
		</AccordionContent>
	</AccordionItem>
</Accordion>
```
