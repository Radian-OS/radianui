## Usage

### Imports

<CodeSnippet title="/components/ui/code-area" code={`import { CodeArea } from "@/components/ui/code-area"`} />

### Code

<CodeSnippet
	title="/components/ui/code-area"
	code={`
<CodeArea 
  code={'const Greeting: React.FC<{ name: string }> = ({ name }) => {\n  return <h1>Hello, {name}!</h1>;\n};'}
  language="tsx"
  theme="github-dark-default"
  lineNumbers={false}
  className="h-full w-full"
/>
`}
/>
