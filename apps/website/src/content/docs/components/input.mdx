## Usage

### Imports

```tsx showLineNumbers title=" "
import {
	Input,
	InputAddon,
	InputGroup,
	InputWrapper,
} from "@/components/ui/input"
```

### Code

```tsx showLineNumbers title=" "
<Input placeholder="Enter username" />

<InputGroup>
    <InputAddon mode="icon">
        <Mail />
    </InputAddon>
    <Input type="email" placeholder="designer@radianui.com" />
</InputGroup>

<InputWrapper>
    <Mail />
    <Input type="email" placeholder="designer@radianui.com" />
</InputWrapper>
```

A flexbox container for grouping an input field together with addons or icons.
