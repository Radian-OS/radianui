## Usage

### Imports

```tsx showLineNumbers title="components/ui/input.tsx"
import {
	Input,
	InputAddon,
	InputGroup,
	InputWrapper,
} from "@/components/ui/input"
```

### Code

```tsx showLineNumbers title="components/ui/input.tsx"
<Input placeholder="Enter username" />

<InputGroup>
    <InputAddon mode="icon">
        <Mail />
    </InputAddon>
    <Input type="email" placeholder="designer@radianos.com" />
</InputGroup>

<InputWrapper>
    <Mail />
    <Input type="email" placeholder="designer@radianos.com" />
</InputWrapper>
```
