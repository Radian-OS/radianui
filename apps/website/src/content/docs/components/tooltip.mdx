## Usage

### Imports

```tsx showLineNumbers
import {
	Tooltip,
	TooltipContent,
	TooltipTrigger,
} from "@/components/ui/tooltip"
```

### Code

```tsx showLineNumbers
<Tooltip>
	<TooltipTrigger>Hover Me</TooltipTrigger>
	<TooltipContent>This is a tooltip.</TooltipContent>
</Tooltip>
```
