## Usage

### Imports

```tsx showLineNumbers title=" "
import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group"
```

### Code

```tsx showLineNumbers title=" "
<RadioGroup defaultValue="apple">
	<RadioGroupItem value="apple" />
	<RadioGroupItem value="banana" />
	<RadioGroupItem value="orange" />
</RadioGroup>
```
