## Usage

### Imports

<CodeSnippet
	title="/components/ui/hover-card.tsx"
	code={`import { 
HoverCard, 
HoverCardTrigger, 
HoverCardContent 
} from "@/components/ui/hover-card.tsx"`}
/>

### Code

<CodeSnippet
	title="/components/ui/hover-card.tsx"
	code={` <HoverCard>
    	<HoverCardTrigger>Hover Me</HoverCardTrigger>

			Hello! I am a tooltip-like card.
		</HoverCardContent>
</HoverCard>`}
/>
