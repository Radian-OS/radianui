## Usage

### Imports

<CodeSnippet title="/components/ui/hover-card.tsx" code={`import {HoverCard} from "@/components/ui/hover-card.tsx"`} />

### Code

<CodeSnippet title="/components/ui/hover-card.tsx" code={` <HoverCard withArrow>

    	<HoverCardTrigger asChild>
    		<button className="px-2 py-1 border rounded">Hover me</button>
    	</HoverCardTrigger>

    	<HoverCardContent>
    		<p className="text-sm">Hello! I am a tooltip-like card.</p>
    	</HoverCardContent>

</HoverCard>`} />
