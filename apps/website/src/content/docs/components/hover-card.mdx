## Usage

### Imports

```tsx showLineNumbers title="/components/ui/hover-card.tsx"
import {
	HoverCard,
	HoverCardContent,
	HoverCardTrigger,
} from "@/components/ui/hover-card"
```

### Code

```tsx showLineNumbers title="/components/ui/hover-card.tsx"
<HoverCard>
	<HoverCardTrigger>Hover Me</HoverCardTrigger>
	<HoverCardContent>Hello! I am a tooltip-like card.</HoverCardContent>
</HoverCard>
```
