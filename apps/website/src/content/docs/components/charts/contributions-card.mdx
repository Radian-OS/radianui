## Usage

### Imports

```tsx showLineNumbers title=" "
import * as React from "react"
import { Scatter, ScatterChart, XAxis, YAxis } from "recharts"
import {
	type ChartConfig,
	ChartContainer,
	ChartTooltip,
} from "@/components/ui/chart"
import { cn } from "@/lib/utils"
```

### Chart Configuration

```tsx showLineNumbers title=" "
const chartConfig = {
	contributions: {
		label: "Contributions",
		color: "var(--color-primary)",
	},
} satisfies ChartConfig
```

### 52-Week Contributions Setup

Render the annual activity grid using `ScatterChart` inside `ChartContainer`:

```tsx showLineNumbers title=" "
<ChartContainer config={chartConfig} className="aspect-auto h-[195px] w-full">
	<ScatterChart margin={{ top: 56, right: 10, bottom: 0, left: 0 }}>
		<XAxis
			type="number"
			dataKey="week"
			domain={[-0.5, 51.5]}
			ticks={MONTH_TICKS}
			tickFormatter={(tick) => MONTH_NAMES[tick] ?? ""}
			interval={0}
			tickLine={false}
			axisLine={false}
		/>
		<YAxis
			type="number"
			dataKey="dayIndex"
			domain={[-0.5, 6.5]}
			ticks={[1, 3, 5]}
			tickFormatter={(val) =>
				val === 1 ? "Mon" : val === 3 ? "Wed" : val === 5 ? "Fri" : ""
			}
			width={36}
			interval={0}
			tickLine={false}
			axisLine={false}
			reversed
		/>
		<ChartTooltip
			cursor={false}
			offset={0}
			isAnimationActive={false}
			animationDuration={0}
			allowEscapeViewBox={{ x: true, y: true }}
			content={({ active, payload }) => {
				if (!active || !payload?.length) return null
				const item = payload[0]?.payload
				return (
					<div className="border-border bg-bg pointer-events-none -translate-x-1/2 -translate-y-[calc(100%+8px)] min-w-36 rounded-lg border px-2.5 py-1.5 text-xs shadow-xl">
						<span className="font-semibold text-fg">
							{item.count > 0
								? `${item.count} contributions`
								: "No contributions"}
						</span>
						<span className="text-fg-tertiary text-[11px]">
							{new Date(item.date).toLocaleDateString("en-US", {
								weekday: "short",
								month: "short",
								day: "numeric",
								year: "numeric",
							})}
						</span>
					</div>
				)
			}}
		/>
		<Scatter
			data={data}
			shape={renderContributionCell}
			isAnimationActive={false}
		/>
	</ScatterChart>
</ChartContainer>
```
