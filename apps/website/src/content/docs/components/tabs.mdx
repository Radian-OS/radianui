## Installation

<Tabs defaultValue="cli">
	<TabsList>
		<TabsTrigger value="cli">CLI</TabsTrigger>
		<TabsTrigger value="manual">Manual</TabsTrigger>
	</TabsList>
	<TabsContent value="cli" className="p-0">
		<PackageManagerTabs
			commands={{
				pnpm: "pnpm dlx @radianos/radianbeta add tabs",
				npm: "npx @radianos/radianbeta add tabs",
				yarn: "yarn @radianos/radianbeta add tabs",
				bun: "bunx @radianos/radianbeta add tabs",
			}}
		/>
	</TabsContent>
	<TabsContent value="manual" className="p-0">
	<Steps>
		<Step>Install the following dependencies:</Step>
		<PackageManagerTabs
			commands={{
				pnpm: "pnpm add @radix-ui/react-tabs @radix-ui/react-slot",
				npm: "npm install @radix-ui/react-tabs @radix-ui/react-slot",
				yarn: "yarn add @radix-ui/react-tabs @radix-ui/react-slot",
				bun: "bun add @radix-ui/react-tabs @radix-ui/react-slot",
			}}
		/>

    	<Step>Copy and paste the following code into your project.</Step>
    	<Tabs defaultValue="tabs">
    		<TabsList>
    			<TabsTrigger value="tabs">Tabs</TabsTrigger>
    			<TabsTrigger value="badge">Badge</TabsTrigger>
    		</TabsList>
    		<TabsContent value="tabs">
    			<ComponentSource name="tabs" title="components/ui/tabs.tsx" collapsible />
    		</TabsContent>
    		<TabsContent value="badge">
    			<ComponentSource name="badge" title="components/ui/badge.tsx" collapsible />
    		</TabsContent>
    	</Tabs>

    	<Step>Update the import paths as per your project structure.</Step>
    </Steps>
    </TabsContent>

</Tabs>

## Usage

### Imports

<CodeSnippet title="/components/ui/tabs" code={`import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs"`} />

### Code

<CodeSnippet
	title="Usage"
	code={`<Tabs defaultValue="account" className="w-[400px]">
		<TabsList>
			<TabsTrigger value="account">Account</TabsTrigger>
			<TabsTrigger value="password">Password</TabsTrigger>
		</TabsList>
		<TabsContent value="account">Make changes to your account here.</TabsContent>
		<TabsContent value="password">Change your password here.</TabsContent>
</Tabs>`}
/>
