## Usage

### Imports

<CodeSnippet title="/components/ui/select" code={`import { Select, SelectTrigger, SelectContent, SelectValue, SelectItem } from "@/components/ui/select"`} />

### Code

<CodeSnippet
	title="/components/ui/select"
	code={`
<Select>
	<SelectTrigger>
		<SelectValue/>
	</SelectTrigger>
	<SelectContent>
		<SelectItem value="male">Male</SelectItem>
		<SelectItem value="female">Female</SelectItem>
	</SelectContent>
</Select>`}
/>
