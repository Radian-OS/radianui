## Usage

### Imports

```tsx showLineNumbers title="/components/ui/select.tsx"
import {
	Select,
	SelectContent,
	SelectItem,
	SelectTrigger,
	SelectValue,
} from "@/components/ui/select"
```

### Code

```tsx
<Select>
	<SelectTrigger>
		<SelectValue />
	</SelectTrigger>
	<SelectContent>
		<SelectItem value="male">Male</SelectItem>
		<SelectItem value="female">Female</SelectItem>
	</SelectContent>
</Select>
```
