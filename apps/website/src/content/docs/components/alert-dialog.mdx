## Usage

### Imports

<CodeSnippet
	title="/components/ui/alert-dialog.tsx"
	code={`import { 
		AlertDialog,
	  	AlertDialogTrigger,
	  	AlertDialogHeader,
	  	AlertDialogTitle,
	  	AlertDialogContent,
	  	AlertDialogDescription,
	  	AlertDialogFooter,
	  	AlertDialogAction,
	  	AlertDialogCancel
} from "@/components/ui/alert-dialog"`}
/>

### Code

<CodeSnippet
	title="/components/ui/alert-dialog.tsx"
	code={`<AlertDialog>
		<AlertDialogTrigger>Show Dialog</AlertDialogTrigger>
		<AlertDialogContent>
			<AlertDialogHeader>
				<AlertDialogTitle>Are you sure?</AlertDialogTitle>
				<AlertDialogDescription>Take a moment to review the details provided to ensure you understand the implications.</AlertDialogDescription>
			</AlertDialogHeader>
			<AlertDialogFooter>
				<AlertDialogCancel>Cancel</AlertDialogCancel>
				<AlertDialogAction>Continue</AlertDialogAction>
			</AlertDialogFooter>
		</AlertDialogContent>
	</AlertDialog>`}
/>
