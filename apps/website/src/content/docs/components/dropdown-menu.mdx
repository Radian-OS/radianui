## Usage

### Imports

```tsx showLineNumbers title=" "
import {
	DropdownMenu,
	DropdownMenuContent,
	DropdownMenuItem,
	DropdownMenuSub,
	DropdownMenuSubContent,
	DropdownMenuSubTrigger,
	DropdownMenuTrigger,
} from "@/components/ui/dropdown-menu"
```

### Code

```tsx showLineNumbers title=" "
<DropdownMenu>
	<DropdownMenuTrigger asChild>
		<Button variant="outline">Open Menu</Button>
	</DropdownMenuTrigger>
	<DropdownMenuContent>
		<DropdownMenuGroup title="Actions" selectionMode="single">
			<DropdownMenuItem value="new">New File</DropdownMenuItem>
			<DropdownMenuItem value="open">Open File</DropdownMenuItem>
			<DropdownMenuItem value="save">Save</DropdownMenuItem>
		</DropdownMenuGroup>
	</DropdownMenuContent>
</DropdownMenu>
```
