## Usage

### Imports

<CodeSnippet
	title="/components/ui/form.tsx"
	code={`import {
Form, 
FormControl, 
FormDescription, 
FormField, 
FormItem, 
FormLabel, 
FormMessage
} from '@/components/ui/form'`}
/>

### Code

<CodeSnippet
	title="/components/ui/form.tsx"
	code={`<Form>
	<FormField
		render={({ field }) => (
			<FormItem>
				<FormLabel>Email</FormLabel>
				<FormControl>
					<Input placeholder="Email address" {...field} />
				</FormControl>
				<FormDescription>Enter your email</FormDescription>
				<FormMessage />
			</FormItem>
		)} 
	/>
</Form>`}
/>
