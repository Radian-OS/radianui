## Usage

### Imports

```tsx showLineNumbers title="/components/ui/button.tsx"
import {
	Button,
	ButtonGroup,
	CompactButton,
	IconButton,
} from "@/components/ui/button"
```

### Code

```tsx showLineNumbers title="/components/ui/button.tsx"
<Button>Click Me</Button>
```
