## Usage

### Imports

<CodeSnippet
	title="components/ui/dropdown.tsx"
	code={`import { Dropdown, DropdownTrigger, DropdownContent, DropdownItem, DropdownSub, DropdownSubTrigger, DropdownSubContent } from "@/components/ui/dropdown"`}
/>

### Code

<CodeSnippet title="components/ui/dropdown.tsx" code={`<Dropdown>

    	<DropdownTrigger asChild>
    		<Button variant="outline">Open Menu</Button>
    	</DropdownTrigger>

    	<DropdownContent>
    		<DropdownGroup title="Actions" selectionMode="single">
    		<DropdownItem value="new">New File</DropdownItem>
    		<DropdownItem value="open">Open File</DropdownItem>
    		<DropdownItem value="save">Save</DropdownItem>
    		</DropdownGroup>
    	</DropdownContent>

</Dropdown>`} />
