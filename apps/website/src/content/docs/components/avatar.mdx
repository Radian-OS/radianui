## Usage

### Imports

```tsx showLineNumbers title="/components/ui/avatar.tsx"
import {
	Avatar,
	AvatarFallback,
	AvatarImage,
	AvatarIndicator,
	AvatarStatus,
} from "@/components/ui/avatar"
```

### Code

```tsx showLineNumbers title="/components/ui/avatar.tsx"
<Avatar size="48" rounded="circle">
	<AvatarImage src="https://radianos.com/logo.png" />
	<AvatarFallback>RS</AvatarFallback>
	<AvatarIndicator>
		<AvatarStatus variant="online" />
	</AvatarIndicator>
</Avatar>
```
