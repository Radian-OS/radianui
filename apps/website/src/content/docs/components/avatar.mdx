## Usage

### Imports

<CodeSnippet
	title="/components/ui/avatar.tsx"
	language="tsx"
	code={`import { 
Avatar,
AvatarFallback,
AvatarImage,
AvatarIndicator,
AvatarStatus,
} from "@/components/ui/avatar"`}
	showLineNumbers
/>

### Code

<CodeSnippet
	title="/components/ui/avatar.tsx"
	code={`<Avatar size="48" rounded="circle">
	<AvatarImage src="https://radianos.com/logo.png" />
	<AvatarFallback>RS</AvatarFallback>
	<AvatarIndicator>
		<AvatarStatus variant="online" />
	</AvatarIndicator>
</Avatar>`}
	showLineNumbers
/>
