## Usage

### Imports

```tsx showLineNumbers title="/components/ui/badge.tsx"
import { Badge, BadgeDot } from "@/components/ui/badge"
```

### Code

```tsx showLineNumbers title="/components/ui/badge.tsx"
<Badge variant="strong" color="success">
	Active
</Badge>
```
