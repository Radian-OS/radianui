## Usage

### Imports

<CodeSnippet title="/components/ui/alert.tsx" code={`import { Alert, AlertActions, AlertContent, AlertDescription, AlertIcon, AlertTitle } from "@/components/ui/alert"`} />

### Code

<CodeSnippet
	title="/components/ui/alert.tsx"
	code={`<Alert>
<AlertIcon><Settings/></AlertIcon>
<AlertTitle>
 Settings Updated
</AlertTitle>
<AlertDescription>
 Your settings have been successfully updated.
</AlertDescription>
<AlertAction>
 <Button variant="strong" color="primary">Action</Button>
</AlertAction>
</Alert>`}
/>
